import gradio as gr
from transformers import AutoTokenizer, AutoModelForSeq2SeqLM

# Load one multilingual model

# Translation function
def translate_with_gradio(text, language_name):
    lang_code = [code for code, name in lang_dict.items() if name == language_name][0]

    tokenizer.src_lang = "en"
    encoded = tokenizer(text, return_tensors="pt")
    generated_tokens = model.generate(**encoded, forced_bos_token_id=tokenizer.get_lang_id(lang_code))
    translated_text = tokenizer.batch_decode(generated_tokens, skip_special_tokens=True)[0]
    return translated_text

# Gradio UI
iface = gr.Interface(
    fn=translate_with_gradio,
    inputs=[
        gr.Textbox(lines=5, label="Input English Text"),
        gr.Dropdown(list(lang_dict.values()), label="Select Target Language")
    ],
    outputs="text",
    title="Universal Multilingual Translator 🌍",
    description="Translate English text into 50+ languages using facebook/m2m100."
)

iface.launch()
